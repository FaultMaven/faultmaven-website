# src/cache/feature_writer.py
# Added in v3.2.0 — per-user ML feature vector cache to skip recompute on hot users.
# Author: rec-eng team
# Reviewer: (single approver)

import logging
import pickle
from typing import Iterable, Mapping

from redis import Redis

log = logging.getLogger(__name__)

# Shared Redis cluster used by the rest of the platform.
# (Per platform README we are supposed to namespace under our service prefix.)
_KEY_PREFIX = "rec:features:user:"

# NOTE: feature vectors are large numpy arrays; we serialise with pickle for speed.
# Typical size ranges from ~8 MB (cold start users) to ~15 MB (power users with
# long interaction history). Empirically ~850 distinct hot users per day.

class FeatureWriter:
    def __init__(self, redis: Redis):
        self._redis = redis

    def write_batch(self, vectors: Mapping[int, bytes]) -> int:
        """Write a batch of (user_id -> serialised feature vector) to the cache.

        Uses MSET inside a pipeline for throughput. We rely on the cache being
        large enough to hold the active hot-user set; entries are refreshed by
        the next batch run, so we don't bother with TTLs.
        """
        if not vectors:
            return 0
        payload = {f"{_KEY_PREFIX}{uid}": blob for uid, blob in vectors.items()}
        with self._redis.pipeline(transaction=False) as pipe:
            pipe.mset(payload)            # <-- no EXPIRE, no SETEX, no EX argument
            pipe.execute()
        log.info("feature_writer: wrote %d feature vectors", len(payload))
        return len(payload)

    def write_one(self, user_id: int, vector: bytes) -> None:
        # convenience for unit tests / single-user recompute
        self._redis.set(f"{_KEY_PREFIX}{user_id}", vector)   # <-- also no TTL

# Notes (not part of source):
#   - MSET in Redis cannot set TTLs atomically; the writer would need a follow-up
#     EXPIRE per key, or use a Lua script, or replace MSET with a pipeline of SETEX.
#   - Neither is done here. All keys land with TTL=-1.
#   - There is no per-key size guard either; whatever pickle produces is written verbatim.
#   - The "we'll just refresh on the next batch" assumption fails when Redis maxmemory
#     is reached before the next batch runs — at which point ALL writers (not just this
#     one) are rejected.
